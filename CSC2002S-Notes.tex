\subsection{Regular Languages}
\subsubsection{Finite automata}
\begin{definition}[Finite automata]
A \textbf{finite automaton} is a 5-tuple $(Q,\Sigma,\delta,q_0,F)$ where:
\begin{itemize}
    \item $Q$ is a finite set of all possible states.
    \item $\Sigma$ is a finite set of all alphabet symbols.
    \item $\delta$ is a transition function $\delta:Q\times\Sigma\to Q$. In other words,
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.3\linewidth]{Pictures/finite_auto_2.png}
    \end{figure}
    \item $q_0$ is the start state.
    \item $F$ is the set of accept states. 
\end{itemize}
\end{definition}

$\quad$\\[-1em]
\textbf{Example:}\\
Let's consider the following finite automaton.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.4\linewidth]{Pictures/finite_auto_1.png}
\end{figure}
$\quad$\\[-2em]
Here, $M_1=(Q,\Sigma,\delta,q_1,F)$ is a finite automaton. For sake of example, here are the exact details of $M_1$.
\begin{itemize}
    \item $Q=\{q_1, q_2. q_3\}$
    \item $\Sigma=\{0, 1\}$
    \item $F=\{q_3\}$
\end{itemize}
It accepts a finite string as input. The output is either \textit{ACCEPT} or \textit{REJECT}.
The computation goes as follows:
\begin{enumerate}
    \item Begin at start state.
    \item Read input symbols.
    \item Follow corresponding transitions.
    \item \textit{ACCEPT} if we reach the accept state, otherwise \textit{REJECT}.
\end{enumerate}
For example: 01101 $\to$ \textit{ACCEPT}, while 00101 $\to$ \textit{REJECT}. Here, $M_1$ accepts all string in $A$ where $A=\{w|w\text{ contains substring }11\}$.
We say that \textbf{$A$ is the language of $M_1 \iff M_1$ recognizes $A \iff A=L(M_1)$}.

\subsubsection{Strings and languages}
A \textbf{string} is a finite sequence of symbols $w_1w_2\cdots w_n\in\Sigma$. A \textbf{language} is a set of string (finite or infinite).
The \textbf{empty string} $\varepsilon$ is the string of length 0. The \textbf{empty language} $\emptyset$ is the set with no strings.
\begin{definition}[Accepting strings]
A finite automaton $M$ \textbf{accepts string} $w=w_1w_2\cdots w_n$ for each $w_i\in\Sigma$ 
if there exists a sequence of states $r_0,r_1,\dots,r_n\in Q$ where:
\begin{itemize}
    \item $r_0=q_0$
    \item $r_i=\delta(r_{i-1},w_i)$ for $1\leq i\leq n$
    \item $r_n\in F$
\end{itemize}
\end{definition}
\begin{definition}[Regular language]
We say a language is $\textbf{regular}$ if some finite automaton recognizes it.
\end{definition}
$\quad$\\[-1em]
\textbf{Example 1:}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.4\linewidth]{Pictures/finite_auto_1.png}
\end{figure}
$\quad$\\[-1em]
$L=(M_1)=\{w|w\text{ contains substring }11\}=A$. Therefore, $A$ is regular.
\\\\
\textbf{Example 2:}\\
Let $B=\{w|w\text{ has an even number of 1s}\}$. In this case, $B$ is regular. (not shown)
\\\\
\textbf{Example 3:}\\
Let $C=\{w|w\text{ has equal number of 0s and 1s}\}$. In this case, $C$ is not regular. (provable)

\subsubsection{Regular expressions}
\begin{definition}[Regular operations]
Let $A,B$ be languages.
\begin{itemize}
    \item \textbf{Union:} $A\cup B=\{w|w\in A\text{ or }w\in B\}$.
    \item \textbf{Concatenation:} $A\circ B=\{xy|x\in A \text{ and }y\in B\}=AB$.
    \item \textbf{Star:} $A^{\ast}=\{x_1\dots x_k|x_i\in A\text{ for }k\geq 0\}$. (Note: $\varepsilon\in A^{\ast}$)
\end{itemize}
\end{definition}
$\quad$\\[-1em]
Regular expressions are built from $\Sigma,\emptyset,\varepsilon$ by using regular operations $\cup,\circ,\ast$.
\\\\
\textbf{Example 1:}
\begin{itemize}
    \item $(0\cup 1)^{\ast}=\Sigma^{\ast}$ gives all strings over $\Sigma$.
    \item $\Sigma^{\ast}1$ gives all strings that end with 1.
    \item $\Sigma^{\ast}11\Sigma^{\ast}$ gives all strings that contain $11=L(M_1)$.
\end{itemize}
\textbf{Example 2:}\\
Let $A=\{\text{good, bad}\}$ and $B=\{\text{boy, girl}\}$.
\begin{itemize}
    \item $A\cup B=\{\text{good, bad, boy, girl}\}$.
    \item $A\circ B=AB=\{\text{goodboy, goodgirl, badboy, badgirl}\}$.
    \item $A^{\ast}=\{\varepsilon, \text{good, bad, goodgood, goodbad, badgood, badbad, goodgoodgood, }\dots\}$
\end{itemize}

\subsubsection{Nondeterministic finite automata}
\begin{definition}[Nondeterministic finite automata]
A \textbf{nondeterministic finite automata (NFA)} $N$ is a 5-tuple $(Q,\Sigma,\delta,q_0,F)$ where
everything is the exact same as a DFA except $\delta$. Here,
$\delta:Q\times\Sigma_{\varepsilon} \to \mathcal{P}(Q)=\{R\hspace{2px}|\hspace{2px} R\subseteq Q\}$.
\end{definition}
$\quad$\\[-1em]
New features of nondeterminism:
\begin{itemize}
    \item Multiple paths possible (0, 1, or many at each step)
    \item $\varepsilon$-transition is a "free" move without reading input.
    \item Accept input if \textbf{some} path leads to the accept state.
\end{itemize}
Nondeterminism doesn't correspond to a physical machine we can build. However, it if useful mathematically.
Here are some ways to think about it:
\begin{itemize}
    \item \textbf{Computationally:} Fork new parallel thread and accept if any thread leads to an accept state.
    \item \textbf{Mathematically:} Tree with branches. Accept if any branch leads to an accept state.
    \item \textbf{Magical:} Guess at each nondeterministic step which way to go. The machine always makes the right guess that leads to the accept state, if possible.
\end{itemize}

\end{figure}
$\quad$\\[-1em]
Here, we have $\delta(q_1,a)=\{q_1, q_2\}$, but $\delta(q_1,b)=\emptyset$.

\subsubsection{Converting NFAs to DFAs}
\begin{note_special}[Theorem]
If an NFA recognizes $A$, then $A$ is regular.
\\\\
\textbf{Proof:}\\
Let NFA $N=(Q,\Sigma,\delta,q_0,F)$ recognize $A$. We need to construct a DFA $M=(Q',\Sigma,\delta',q_0',F')$, ignoring the $\varepsilon$-transitions.
Here $M$ is defined as:
\begin{itemize}
    \item $Q'=\mathcal{P}(Q)$.
    \item $\delta'(R\in Q',a)=q\in\delta(r,a)$ for some $r\in R$.
    \item $q_0'=\{q_0\}$.
    \item $F'=\{R\in Q'|R\cap F\}$.
\end{itemize}
\vspace{-2em}

    \centering
    \includegraphics[width=0.5\linewidth]{Pictures/nfa_2.png}
\end{figure}
\end{note_special}

\subsubsection{Closure properties for regular languages}
\begin{note_special}[Theorem: Closure under $\cup$.]
If $A_1$ and $A_2$ are regular languages, then so is $A_1\cup A_2$.
\\\\
\textbf{Proof:}\\
Let $M_1=(Q_1,\Sigma,\delta_1,q_1,F_1)$ recognize $A_1$ and similarly, let $M_2=(Q_2,\Sigma,\delta_2,q_2,F_2)$ recognize $A_2$.
We need to construct $M=(Q,\Sigma,\delta,q_0,F)$ where
\begin{itemize}
    \item $Q=Q_1\times Q_2=\{(q_1,q_2)\hspace{2px}|\hspace{2px}q_1\in Q_1\wedge q_2\in Q_2\}$.
    \item $q_0=(q_1,q_2)$.
    \item $\delta((q,r),a)=\big(\delta_1(q,a),\delta_2(q,a)\big)$.
    \item $F=(F_1\times Q_2)\times(Q_1\times F_2)$.
\end{itemize}

\end{figure}
\end{note_special}

Every GNFA $G$ has an equivalent regular expressions $R$.
\\\\
\textbf{Proof:}\\
We will make use of an inductive argument on the number of states $k$ of $G$. For the base step ($k=2$):
\begin{figure}[H]

    \includegraphics[width=0.4\linewidth]{Pictures/gnfa_2.png}

Assume the Lemma holds true for $k-1$ states. Now we want to prove it holds for $k$ states.

\subsubsection{Non-regular languages}
How do we show a language is not regular?
\begin{itemize}
    \item Remember, to show a language is \textit{regular}, we give an equivalent DFA.
    \item To show a language is \textit{not regular}, we must give a proof.
    \item It is not enough to say you couldn't find a DFA for it, therefore it isn't regular!
\end{itemize}
\textbf{Example 1:}\\
Let $B=\{w|w\text{ has equal numbers of 0s and 1s}\}$. One's intuition syas that $B$ is not regular since DFAs cannot count unboundedly.
\\\\
\textbf{Example 2:}\\
Let $C=\{w|w\text{ has equal numbers of 01 and 10 substrings}\}$. As examples, $0101\not\in C$ and $0110\in C$.
Again, one's intuition says that $C$ is not regular since DFA cannot count unboundedly.
However, $C$ is regular!

\subsubsection{Proving non-regularity}
\begin{note_special}[Pumping Lemma]
For every regular language $A$, there exists a number $p$, called the "pumping length", such that if $s\in A$ and $|s|\ge p$ then $s=xyz$ where
\begin{enumerate}
    \item $xy^{i}z\in A$ for all $i\ge 0\quad$ ($y^{i}=\underbrace{yy\cdots y}_i$).
    \item $y\ne\varepsilon$.
    \item $|xy|\le p$.
\end{enumerate}
Informally, if $A$ is regular, then every long string in $A$ can be "pumped" and the result stays in $A$.
\\\\
\textbf{Proof:}\\
Let DFA $M$ recognize $A$. Let $p$ be the number of states in $M$. Pick $s\in A$ where $|s|\ge p$.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.4\linewidth]{Pictures/nonr_1.png}
\end{figure}

$M$ will repeat a state $q_j$ when reading $s$ because $s$ is so long.
The path that $M$ follows when reading $s$.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.4\linewidth]{Pictures/nonr_2.png}
\end{figure}
\end{note_special}
$\quad$\\[-1em]
\textbf{Exmaple 1:}\\
Let $D=\{0^k1^k|k\ge 0\}$. Show that $D$ is not regular.
\\\\
Assume, for the sake of contradiction, that $D$ is regular. \\
Since $D$ is regular, the pumping lemma must hold. Let $s=0^p1^p\in D$. \\
The pumping lemma says that we can divide $s=xyz$. \\
But $xyyz$ has excess 0s, and thus $xyyz\not\in D$ contradicting the pumping lemma. \\
Therefore, our assumption is false and thus $D$ is not regular.

$\quad$\\[-1em]
\textbf{Example 2:}\\
Let $F=\{ww|w\in\Sigma^{\ast}\}$. Say $\Sigma^{\ast}=\{0,1\}$. Show that $F$ is not regular.
\\\\
Assume, for the sake of contradiction, that $F$ is regular. \\
Since $F$ is regular, the pumping lemma must hold. \\
We need to choose $s\in F$. Let $s=0^p10^p1\in F$. \\
We can show that $s=xyz$ cannot be pumped satisfying the 3 conditions. \\
Thus, $xyyz\not\in F$, which is a contradiction. Therefore, $F$ is not regular.
\\\\
\textbf{Example 3:}\\
Let $B=\{w|w\text{ has equal numbers of 0s and 1s}\}$. Show that $B$ is not regular.
\\\\
Assume, for the sake of contradiction, that $B$ is regular. \\
We know $0^{\ast}1^{\ast}$ is regular, and so $B\cap 0^{\ast}1^{\ast}$ is regular (closure under intersection). \\
But $D=B\cap 0^{\ast}1^{\ast}$ and we already showed $D$ is not regular (from example 1). \\
Therefore, $B$ is not regular.
\\\\
\textbf{Example 4:}\\
Let $A=\{1^{n^2}|n\ge 0\}$. \textit{($A$ contains all strings of 1s whose length is a perfect square)}. Show that $A$ is not regular.
\\\\
Assume, for the sake of contradiction, that $A$ is regular.\\
Let $p$ be the pumping length given by the pumping lemma. Take $s=1^{p^2}\in A$.\\
The pumping lemma guarantees $s=xyz$ with $s=xy^iz\in A$ for some $i\ge 0$.\\
Consider the two strings $xyz$ and $xy^2z$. The length of these strings differ by $y$.\\
By contition 3, $|xy|\le p$ and thus $|y|\le p$. Then, we have $|xyz|=p^2$ and so $|xy^2z|\le p^2+p$.\\
But $p^2+p<p^2+2p+1=(p+1)^2$.\\
Moreover, condition 2 implies that $y$ is not the empty string and so $|xy^2z|>p^2$.\\
Therefore, the length of $xy^2z$ lies strictly between the consecutive squares $p^2$ and $(p+1)^2$.\\
Hence this length cannot be a perfect square itself, which is a contradiction.\\
Thus, $xy^2z\not\in A$ and so we conclude that $A$ is not regular.
\\\\
\textbf{Example 5:}\\
Let $E=\{0^i1^j|i>j\}$. Show that $E$ is not regular.
\\\\
Assume, for the sake of contradiction, that $E$ is regular.\\
Let $p$ be the pumping length given by the pumping lemma. Take $s=0^{p+1}1^p\in E$.\\
By condition 1, $xy^iz\in E$ even when $i=0$. Consider the string $xy^0z=xz$.\\
Removing string $y$ decreases the number of 0s in $s$.\\
But, recall that $s$ has just one more 0 than 1.\\
Therefore, $xz$ cannot have more 0s than 1s and so $s=0^{p+1}1^p\not\in E$. \\
Hence a contradiction. Therefore, $E$ is not regular.
\subsection{Context-free languages}
\subsubsection{Context-free grammars}
\begin{definition}[Context-free grammar]
A \textbf{Context Free Grammar (CFG)} $G$ is a 4-tuple $(V,\Sigma,R,S)$ where
\begin{itemize}
    \item $V$ is the finite set of variables.
    \item $\Sigma$ is the finite set of terminal symbols.
    \item $R$ is the finite set of rules (rule form: $V\to (V\cup \Sigma)^{\ast}$).
    \item $S$ the starting variable.
\end{itemize}
For $u,v\in(V\cup\Sigma)^{\ast}$ write
\begin{enumerate}
    \item $u\Rightarrow v$ if you can go from $u$ to $v$ with one substitution step in $G$.
    \item $u\overset{\ast}{\Rightarrow} v$ if you can go from $u$ to $v$ with some number of substitution steps in $G$. \\
    $u\Rightarrow u_1\Rightarrow u_2\Rightarrow \cdots\Rightarrow u_k\Rightarrow v$ is called a \textit{derivation} of $v$ from $u$.
    Furthermore, if $u=S$ then it is a \textit{derivation} of $v$.
\end{enumerate}
$L(G)=\{w|w\in\Sigma^{\ast}\text{ and }S\overset{\ast}{\Rightarrow} w\}$.
\end{definition}
\begin{definition}[Context-free language]
$A$ is a \textbf{Context Free Language (CFL)} is $A=L(G)$ for some CFG $G$.
\end{definition}

\subsubsection{Ambiguity}
Take the following two CFGs:\\
\begin{minipage}{0.5\textwidth}
\begin{flalign*}
G_1& &\\
E&\rightarrow E+T\hspace{4px}|\hspace{4px} T &\\
T&\rightarrow T\times F\hspace{4px}|\hspace{4px} F &\\
F&\rightarrow (E)\hspace{4px}|\hspace{4px} a
\end{flalign*}
\end{minipage}
\begin{minipage}{0.5\textwidth}
\begin{flalign*}
G_2& &\\
E&\rightarrow E+E\hspace{4px}|\hspace{4px}E\times E\hspace{4px}|\hspace{4px}(E)\hspace{4px}|\hspace{4px}a
\end{flalign*}
\end{minipage}
$\quad$\\\\[-1em]
Both $G_1$ and $G_2$ recognize the same language, i.e. $L(G_1)=L(G_2)$.
However, $G_1$ is an unambiguous CFG and $G_2$ is ambiguous.

\subsubsection{Pushdown automaton}
\begin{definition}[Pushdown automaton]
A \textbf{Pushdown Automaton (PDA)} is a 6-tuple $(Q,\Sigma,\Gamma,\delta,q_0,F)$:
\begin{itemize}
    \item $\Sigma$ is the input alphabet.
    \item $\Gamma$ is the stack alphabet.
    \item $\delta:Q\times \Sigma_{\varepsilon}\times\Gamma_{\varepsilon}\rightarrow\mathcal{P}(Q\times\Gamma_{\varepsilon})$\\
    $\delta(q,a,c)=\{(r_1,d),(r_2,e)\}$
\end{itemize}
We accept if some thread is in the accept state at the end of the input string.
\end{definition}
$\quad$\\[-1em]
A PDA operates like an NFA except can write/add (push) or read/remove (pop) symbols from the top of the stack.
The input appears on a tape.
\\\\
\textbf{Example 1:}\\
PDA for $D=\{0^k1^k:k\ge 0\}$
\begin{enumerate}
    \item Read 0s from input, push 0 onto the stack until a 1 is read.
    \item Read 1s from input, while popping 0s from the stack.
    \item Enter the accept state if the stack is empty.
\end{enumerate}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\linewidth]{Pictures/pda_1.png}
\end{figure}
\begin{note}[How to read transitions for PDAs]
\textbf{Example 1:} Take the transition $\varepsilon,\hspace{2px}\varepsilon;\hspace{2px}\$$. 
\begin{itemize}
    \item The first $\varepsilon$ is the input read from the tape. Since we're reading $\varepsilon$, this is equivalent to a blank on the tape.
    \item The second $\varepsilon$ is the value on the top of the stack. Since we're reading $\varepsilon$, we assume the stack is empty.
    \item The last $\$$ is part of the stack alphabet. This signifies the bottom of the stack. We want to "push" this value onto the stack.
\end{itemize}
\textbf{Example 2:} Take the transition $\varepsilon,\hspace{2px}\$;\hspace{2px}\varepsilon$. 
\begin{itemize}
    \item The first $\varepsilon$ is the input read from the tape. Since we're reading $\varepsilon$, this is equivalent to a blank on the tape.
    \item The $\$$ is the value we expect on the top of the stack. 
    \item The second $\varepsilon$ is the value we push onto the stack. We're effectively replacing the bottom-of-stack character with a blank, clearing the stack.
\end{itemize}
\end{note}
\textbf{Example 2:}\\
PDA for $B=\{ww^{\mathcal{R}}:w\in\{0,1\}^{\ast}\}$ with sample input: 011110.
\begin{enumerate}
    \item Read and push input symbols onto the stack. Non-deterministically either repeat or go to (2).
    \item Read input symbols and pop stack symbols, and compare. If they are ever not equal, then the thread rejects.
    \item Enter the accept state if the stack is empty.
\end{enumerate}

\subsubsection{Converting CFGs to PDAs}
\begin{note_special}[Theorem]
If $A$ is a CFG, then some PDA recognizes $A$.
\\\\
\textbf{Proof:}\\
Convert $A$'s CFG to a PDA.
\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\linewidth]{Pictures/pda_2.png}
\end{figure}
\begin{enumerate}
    \item Push the start symbol onto the stack.
    \item If the top of the stack is:
    \begin{itemize}
        \item \textit{Variable:} replace with right hand side of rule (non-deterministic choice).
        \item \textit{Terminal:} pop it  and match with next input symbol.
    \end{itemize}
    \item If the stack is empty, enter the accept state.
\end{enumerate}
\end{note_special}

\subsubsection{Equivalence of CFGs and PDAs}
\begin{note_special}[Theorem]
$A$ is a CFL if and only if some PDA recognizes $A$. \textit{(Proof not examinable)}
\end{note_special}
\begin{note_special}[Corollaries]
\begin{enumerate}
    \item Every regular language is a CFL.
    \item If $A$ is a CFL and $B$ is regular, then $A\cap B$ is a CFL.
\end{enumerate}
\textbf{Note 1:} If $A$ and $B$ are CFLs then $A\cap B$ may not be a CFL. Therefore, the class of CFLs is not closed under $\cap$.
\\
\textbf{Note 2:} The class of CFL is closed under $\cup,\circ,\ast$.
\end{note_special}

\subsubsection{Chomsky normal form (CNF)}
Chomsky normal form is a simplification of grammars:
\begin{itemize}
    \item If the start symbol is recursive, remove it.
    \item Removing $\varepsilon$-productions.
    \item Removing unit productions.
    \item Removing "useless" productions.
\end{itemize}
\begin{note_special}[Converting to CNF]
Given a grammar $G=\{V,T,S,P\}$, we can derive $G'=\{V,T,S',P'\}$ as follows:
\\\\
\textbf{Removing recursive start symbol:}\\
Introduce a new start symbol $S'$. Let $P'$ be the production rule set $P$ of $G$ with the following rule added to it: $S'\rightarrow S$. 
\\\\
\textbf{Remove $\varepsilon$-productions:}\\
If there is a production of the form $A\rightarrow\varepsilon$ then $A$ is a nullable variable.
If $B\rightarrow A_1A_2\cdots A_n$ and $A_1A_2\cdots A_n$ are nullable variables, then $B$ is also a nullable variable.
A given grammar $G$ can be converted into a grammar $G'$ that does not contain nullable variables.
\begin{enumerate}
    \item Remove each productions rule of the form $A\rightarrow\varepsilon$.
    \item Add each rule of the form $A\rightarrow s_1s_2\cdots s_n$ with $s_i\in V\cup T$ such that each $s_i$ is not a nullable variable to $P'$.
    \item For each production rule $A\rightarrow s_1N_1s_2N_2\cdots s_nN_n$, where each $s_i$ is not a nullable variable and $N_1,N_2,\dots,N_n$ are nullable variables, add a copy of the rule with one or more $N_i$'s replaced by $\varepsilon$ to $P'$, e.g. $A\rightarrow s_1s_2N_2\cdots s_nN_n$.
    Note that if the $s_i$'s are also nullable, then we do not add $A\rightarrow\varepsilon$ unless $A$ is the start symbol.
\end{enumerate}
\textbf{Remove unit productions:}
\begin{itemize}
    \item Draw a dependency graph with edge $(A,B)$ for each unit production of the form $A\rightarrow B$.
    \item From the graph, identify all $A$ $B$.
    \item For all $A$ and $B$ identified, add a production of the form $A\rightarrow s_1\hspace{4px}|\hspace{4px}s_2\hspace{4px}|\hspace{4px}\cdots\hspace{4px}|\hspace{4px} s_n$ to $P'$ where
    $B\rightarrow \vec{s_1\hspace{4px}|\hspace{4px}s_2\hspace{4px}|\hspace{4px}\cdots\hspace{4px}|\hspace{4px} s_n}$. Note that each $s_i\in V\cup T$ and none of the $s_i$ form unit productions.
\end{itemize}
\textbf{Remove useless productions:}\\
There are essentially two types of useless productions:
\begin{itemize}
    \item Productions containing variables that do not terminate.
    \item Productions containing variables that are not reachable.
\end{itemize}
Identify these productions are remove them from $P'$.
\\\\
\textbf{Correct production rule format:}\\
A context-free grammar $G$ is in CNF, if the production rules have the following form:
\begin{itemize}
    \item $A\rightarrow AB$ where $A$ and $B$ are variables, or
    \item $A\rightarrow a$ where $a$ is a single terminal.
\end{itemize}
\end{note_special}
\textbf{Example:}\\
Take the following grammar:
\begin{flalign*}
S&\rightarrow ASA\hspace{4px}|\hspace{4px}aB &\\
A&\rightarrow B\hspace{4px}|\hspace{4px}S &\\
B&\rightarrow b\hspace{4px}|\hspace{4px}\varepsilon
\end{flalign*}
\textbf{Step 1: Remove recursive start symbol}
\begin{flalign*}
S'&\rightarrow S &\\
S&\rightarrow ASA\hspace{4px}|\hspace{4px}aB &\\
A&\rightarrow B\hspace{4px}|\hspace{4px}S &\\
B&\rightarrow b\hspace{4px}|\hspace{4px}\varepsilon
\end{flalign*}
\textbf{Step 2: Remove $\varepsilon$-productions}
\begin{flalign*}
S'&\rightarrow S &\\
S&\rightarrow ASA\hspace{4px}|\hspace{4px}aB\hspace{4px}|\hspace{4px}AS\hspace{4px}|\hspace{4px}SA\hspace{4px}|\hspace{4px}S\hspace{4px}|\hspace{4px}a &\\
A&\rightarrow B\hspace{4px}|\hspace{4px}S &\\
B&\rightarrow b
\end{flalign*}
\textbf{Step 3: Remove unit productions}
\begin{flalign*}
S'&\rightarrow ASA\hspace{4px}|\hspace{4px}aB\hspace{4px}|\hspace{4px}AS\hspace{4px}|\hspace{4px}SA\hspace{4px}|\hspace{4px}a &\\
S&\rightarrow ASA\hspace{4px}|\hspace{4px}aB\hspace{4px}|\hspace{4px}AS\hspace{4px}|\hspace{4px}SA\hspace{4px}|\hspace{4px}a &\\
A&\rightarrow b\hspace{4px}|\hspace{4px}ASA\hspace{4px}|\hspace{4px}aB\hspace{4px}|\hspace{4px}AS\hspace{4px}|\hspace{4px}SA\hspace{4px}|\hspace{4px}a &\\
B&\rightarrow b
\end{flalign*}
\textbf{Step 4: Ensure correct production rule format}
\begin{flalign*}
S'&\rightarrow AA1\hspace{4px}|\hspace{4px}UB\hspace{4px}|\hspace{4px}AS\hspace{4px}|\hspace{4px}SA\hspace{4px}|\hspace{4px}a &\\
S&\rightarrow AA1\hspace{4px}|\hspace{4px}UB\hspace{4px}|\hspace{4px}AS\hspace{4px}|\hspace{4px}SA\hspace{4px}|\hspace{4px}a &\\
A&\rightarrow b\hspace{4px}|\hspace{4px}AA1\hspace{4px}|\hspace{4px}UB\hspace{4px}|\hspace{4px}AS\hspace{4px}|\hspace{4px}SA\hspace{4px}|\hspace{4px}a &\\
A1&\rightarrow SA &\\
U&\rightarrow a &\\
B&\rightarrow b
\end{flalign*}

\subsubsection{Proving languages are not context free}
\begin{note_special}[Pumping Lemma for CFLs]
For every CFL $A$, there is a $p$ such that if $s\in A$ and $|s|\ge p$ then $s=uvxyz$ where
\begin{enumerate}
    \item $uv^ixy^iz\in A$ for all $i\ge 0$.
    \item $vy\neq\varepsilon$.
    \item $|vxy|\le p$.
\end{enumerate}
Informally, all long string in $A$ are pumpable and stay in $A$.
\end{note_special}
$\quad$\\[-1em]
\textbf{Example 1:}\\
Let $B=\{0^k1^k2^k|k\ge 0\}$. Show that $B$ is not a CFL.
\\\\
Assume, for the sake of contradiction, that $B$ is a CFL.\\
The CFL pumping lemma gives $p$ as above. Let $s=0^p1^p2^p\in B$.\\
The pumping lemma says that we can divide $s=uvxyz$ satisfying the 3 conditions.\\
Condition 3 ($|vxy|\le p$) implies that $vxy$ cannot contain both 0s and 2s.\\
So $uv^2xy^2z\not\in B$, violating condition 1, which is a contradiction.\\
Therefore, our assumption is false. We conclude that $B$ is not a CFL.
\\\\
\textbf{Example 2:}\\
Let $F=\{ww|w\in\Sigma^{\ast}\}$ where $\Sigma=\{0,1\}$. Show that $F$ is not a CFL.
\\\\
Assum, for the sake of contradiction, that $F$ is a CFL.\\
The CFL pumping lemma gives $p$ as above. Take $s=0^p1^p0^p1^p\in F$.\\
Condition 3 implies that $vxy$ does not overlap two runs of 0s or two runs of 1s.\\
Therefore, in $uv^2xy^2z$, two runs of 0s or two runds of 1s have unequal length.\\
So $uv^2xy^2z\not\in F$ violating Condition 1, which is a contradiction.\\
Thus, $F$ is not a CFL.

\subsection{Turing Machines}
\subsubsection{Turing machines}
\begin{definition}[Turing machines]
A \textbf{turing machine} $M$ is a 7-tuple $(Q,\Sigma,\Gamma,\delta,q_0,\square,F)$ where:
\begin{itemize}
    \item $Q$ is the set of states.
    \item $\Sigma$ is the input alphabet \textit{(finite)}.
    \item $\Gamma$ is the tape alphabet \textit{(finite)}.
    \item $\delta$ is the transition function.
    \item $q_0\in Q$ is the initial/start state.
    \item $F\subseteq Q$ is the final state(s).
    \item $\square\in\Gamma$ is the blank symbol.
\end{itemize}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.6\linewidth]{Pictures/tm_1.png}
\end{figure}
\end{definition}
$\quad$\\[-1em]
There some conditions:
\begin{itemize}
    \item $\Sigma\subseteq\Gamma$, since the input is written on the tape initially.
    \item $\square\in\Gamma\setminus\Sigma$, since the rest of the tape is blank.
\end{itemize}
Initially, we're in state $q_0$. The tape contains input $w$ surrounded by $\square$.
The tape head is in the left-most cell of the input also denoted with $D$. The transitions work as follows:
\begin{itemize}
    \item $\delta:Q\times\Gamma\Rightarrow Q\times\Gamma\times\{L,R\}$.
    \item $\delta(q,x)=(p,y,d)$ means that if $M$ is in state $q$ and the tape head is over symbol $x$, then $M$:
    \begin{enumerate}
        \item Changes state to $p$.
        \item Replaces $x$ by $y$ on the tape.
        \item Moves the tape head by one cell in a direction $d$ \textit{(Either left or right)}.
    \end{enumerate}
\end{itemize}

\subsubsection{Determinism}
A turing machine is \textbf{deterministic} if, for each $\delta(q,x)$ we have at most one move \textit{($\delta(q,x)$ could also be undefined)}.
\\\\
\textbf{Acceptance:}\\
$w$ is accepted by a turing machine $M$ if $M$, when started with $w$ on the tape, eventually enters a final state. We can assume that all final states are halting, i.e. no transition is defined for them.
\\\\
\textbf{Rejection:}\\
$M$ halts in non-final states. In other words, in states for which no transition is defined, or it never halts \textit{(Infinite loop)}.

\subsubsection{Instantaneous descriptions}
This is the configuration of a turing machine. It describes the current situation of the turing machine (and the tape).
$$
\text{ID}=\propto q\beta\text{ with }q\in Q,\propto_1,\propto_2\in\Gamma^{\ast}
$$
Meaning:
\begin{itemize}
    \item The non-blank portion of the tape contains $\propto,\beta$.
    \item The head is on the left-most symbol of $\beta$.
    \item The machine is in state $q$.
\end{itemize}
We will use $\vdash$ and $\vdash^{\ast}$ to denote the changes.

\subsubsection{Language accepted by a turing machine}
Let $M=(Q,\Sigma,\Gamma,\delta,q_0,\square,F)$ be a turing machine. Then, the language accepted by $M$ is
$$
L(M)=\{w\in\Sigma^{\ast}|q_0w\vdash^{\ast}\propto p\beta\text{ with }p\in F\text{ and }\propto_1\beta\in\Gamma^{\ast}\}
$$
The set of all languages we can accept using a turing machine is often called the recursively enumerable languages.
Conversely, a turing machine can accept a language by halting:
$$
H(M)=\{w\in\Sigma|q_0q\vdash^{\ast}\propto p\beta\text{ where there is no possible move from}\propto p\beta\}
$$
